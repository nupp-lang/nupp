export function registers(): string
    local reserved: {[integer]: boolean} = {}
    for _, physical in ipairs(target.reserved) do
        assert(not reserved[physical])
        reserved[physical] = true
    end
    local seen: {[integer]: boolean} = {}
    for _, registers in ipairs({target.integerRegisters, target.floatRegisters}) do
        for _, physical in ipairs(registers) do
            assert(not reserved[physical] and not seen[physical])
            seen[physical] = true
        end
    end
    assert(#target.integerRegisters == 21 and #target.floatRegisters == 31)
    assert(target.register(1) == 0 and target.register(32) == 31)
    assert(target.register(65) == 0 and target.register(96) == 31)
    assert(not pcall(target.register, 0) and not pcall(target.register, 33))
    assert(not pcall(target.register, 64) and not pcall(target.register, 97))
    assert(target.CONTEXT == 19 and target.FRAME == 20 and target.DISPATCH == 21)
    assert(target.IMAGE == 22 and target.SPILL == 23)
    assert(reserved[target.CONTEXT + 1] and reserved[target.FRAME + 1])
    assert(reserved[target.DISPATCH + 1] and reserved[target.IMAGE + 1] and reserved[target.SPILL + 1])
    assert(#target.internalIntegerPreserved == 0 and #target.internalFloatPreserved == 0)
    assert(#target.cIntegerPreserved == 11 and #target.cFloatPreserved == 8)
    for _, physical in ipairs(target.cAllocatableIntegerPreserved) do
        assert(seen[physical] and physical >= 25 and physical <= 29)
    end
    assert(target.stackAlignment == 16)
    return "architectural, allocator, pinned and C/internal convention register sets checked"
end
