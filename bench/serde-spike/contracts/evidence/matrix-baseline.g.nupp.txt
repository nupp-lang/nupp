local serde = require("nupp.serde")
local json = require("nupp.codec.json")
local text = require("nupp.text")
@derive(nupp.derive.Serde)
local record Small
    id: uint32
    active: boolean
    name: string
end
@derive(nupp.derive.Serde)
local record Medium
    a: integer
    b: integer
    c: integer
    d: integer
    e: integer
    f: integer
    g: integer
    h: integer
    i: integer
    j: integer
    k: integer
    l: integer
end
@derive(nupp.derive.Serde)
local record Collections
    rows: {Small}
    labels: {[string]: string}
end
@derive(nupp.derive.Serde)
local record Tree
    name: string
    next: Tree?
end
local function operationFor(target)
    local operation = json.newCodec():prepare(serde.of<any>(target))
    return {encode = function(value) return operation:encode(value) end,
        decode = function(bytes) local value, problem = operation:decode(bytes); assert(problem == nil, problem); return assert(value) end,
        append = function(value: any, exclusive buffer: text.Buffer) operation:write(value, buffer) end}
end
local cases = {}
local function add(name, target, input)
    local operation = operationFor(target)
    local value = operation.decode(input)
    local canonical = operation.encode(value)
    assert(operation.encode(operation.decode(canonical)) == canonical)
    local buffer = text.newBuffer()
    cases[#cases + 1] = {name = name, input = input, canonical = canonical,
        encode = function() return operation.encode(value) end,
        decode = function() return operation.decode(input) end,
        buffer = function() buffer:reset(); operation.append(value, buffer); return #buffer end,
        cold = function() return operationFor(target) end}
end
add("small", Small, '{"id":7,"active":true,"name":"sample"}')
add("medium", Medium, '{"a":1,"b":2,"c":3,"d":4,"e":5,"f":6,"g":7,"h":8,"i":9,"j":10,"k":11,"l":12}')
add("reverse", Medium, '{"l":12,"k":11,"j":10,"i":9,"h":8,"g":7,"f":6,"e":5,"d":4,"c":3,"b":2,"a":1}')
local rows = {}
for i = 1, 32 do rows[i] = '{"id":' .. tostring(i) .. ',"active":true,"name":"sample"}' end
add("collections", Collections, '{"rows":[' .. table.concat(rows, ',') .. '],"labels":{"a":"one","b":"two"}}')
local recursionOk, recursionProblem = pcall(add, "recursive", Tree, '{"name":"a","next":{"name":"b","next":{"name":"c","next":null}}}')
add("large-string", Small, '{"id":7,"active":true,"name":' .. json.encode(string.rep('large \" payload ', 4096)) .. '}')
cases.unsupported = not recursionOk and tostring(recursionProblem) or nil
return cases
